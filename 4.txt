const crypto = require('crypto')
class Block{
	constructor(index, time, data, prevhash){
		this.index = index;
		this.time = time;
		this.data = data;
		this.prevhash = prevhash;
		this.hash = this.calchash()
		this.nonce = 0;
	}
	calchash(){
		return crypto.createHash('sha256').update(this.index + this.time + this.data + this.prevhash + this.nonce).digest('hex');
	}
	mine(difficulty){
		let target = "0".repeat(difficulty);
		while(!this.hash.startsWith(target)){
			this.nonce++;
			this.hash = this.calchash();
		}
		console.log("Mined! Nonce: ", this.nonce, " Hash: ", this.hash);
	}
}
class Blockchain{
	constructor(diff){
		this.chain = [new Block(0, Date.now(), "Genesis", "0")];
		this.diff = diff;
	}
	addBlock(data){
		let last = this.chain[this.chain.length-1];
		let block = new Block(last.index+1, Date.now(), data, last.hash);
		block.mine(this.diff);
		this.chain.push(block);
	}
}
let bc = new Blockchain(4);
bc.addBlock("Alice spent 5 ETH");
bc.addBlock("Ron spent 1 ETH");
