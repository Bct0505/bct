const crypto = require("crypto");
const express = require("express");
const app = express();

class Blockchain {
  constructor() {
    this.chain = [];
    this.pending = [];
    this.createBlock(100, "1");
  }
  createBlock(proof, prevHash) {
    let block = {
      index: this.chain.length + 1,
      time: Date.now(),
      transactions: this.pending,
      proof: proof,
      prevHash: prevHash
    };
    this.pending = [];
    this.chain.push(block);
    return block;
  }
  get last() {
    return this.chain[this.chain.length - 1];
  }
  hash(block) {
    return crypto.createHash("sha256").update(JSON.stringify(block)).digest("hex");
  }
  proofOfWork(lastProof) {
    let proof = 0;
    while (!this.valid(lastProof, proof)) proof++;
    return proof;
  }
  valid(last, proof) {
    let guess = crypto.createHash("sha256").update(String(last) + String(proof)).digest("hex");
    return guess.substring(0, 4) === "0000";
  }
}

let bc = new Blockchain();

app.get("/mine", (req, res) => {
  let last = bc.last;
  let proof = bc.proofOfWork(last.proof);
  bc.pending.push({from: "0", to: "miner", amount: 1});
  let block = bc.createBlock(proof, bc.hash(last));
  res.json({message: "Block mined", block});
});

app.get("/chain", (req, res) => {
  res.json({chain: bc.chain, length: bc.chain.length});
});

app.listen(5000, () => console.log("Server on 5000"));
