import hashlib, json, time, os


def SHA256(data):
    if isinstance(data, str): data = data.encode()
    return hashlib.sha256(data).hexdigest()


class Block:
    def __init__(self, index, time, data, prevhash="0"):
        self.index = index
        self.time = time
        self.data = data
        self.prevhash = prevhash
        self.hash = self.calchash()

    def calchash(self):
        return SHA256(f"{self.index}{self.time}{self.prevhash}{json.dumps(self.data, sort_keys=True)}")


class Blockchain:
    def __init__(self):
        self.chain = [Block(0, time.time(), {"info": "Genesis"}, "0")]

    def addDoc(self, name, data):
        doc_hash = SHA256(data)
        prev = self.chain[-1]
        block = Block(len(self.chain), time.time(),
                      {"name": name, "hash": doc_hash}, prev.hash)
        self.chain.append(block)
        return doc_hash

    def verify(self, data):
        target = SHA256(data)
        for block in self.chain[1:]:
            if block.data.get("hash") == target:
                return True, block.data, block.time
        return False, None, None

    def isValid(self):
        for i in range(1, len(self.chain)):
            cur, prev = self.chain[i], self.chain[i - 1]
            if cur.hash != cur.calchash(): return False
            if cur.prevhash != prev.hash: return False
        return True


def ask(msg):
    v = input(msg).strip()
    if v.lower() == "exit": raise SystemExit
    return v


def main():
    chain = Blockchain()
    chain.addDoc("Invoice", "Invoice #10042 - Total Due: $1,500.00 - Vendor: Acme Corp")

    while True:
        c = ask("1) Register text  2) Verify text  3) Register file  4) Verify file  5) Audit: ")

        if c in "12":
            content = ask("Text: ")
            if c == "1":
                name = ask("Name: ")
                print("Hash:", chain.addDoc(name, content))
            else:
                ok, meta, ts = chain.verify(content)
                print("VALID" if ok else "INVALID")
                if ok: print(meta["name"], time.ctime(ts))

        elif c in "34":
            path = ask("File path: ").strip('"')
            if not os.path.exists(path):
                print("Not found")
                continue
            data = open(path, "rb").read()
            if c == "3":
                print("Hash:", chain.addDoc(os.path.basename(path), data))
            else:
                ok, meta, ts = chain.verify(data)
                print("VALID" if ok else "INVALID")
                if ok: print(meta["name"], time.ctime(ts))

        elif c == "5":
            print("SECURE" if chain.isValid() else "TAMPERED", "-", len(chain.chain), "blocks")

        else:
            print("Invalid")


if __name__ == "__main__":
    main()
