const { Web3 } = require('web3');
const PUBLIC_RPC = 'https://ethereum-sepolia-rpc.publicnode.com';
const web3 = new Web3(PUBLIC_RPC);

const TEACHER_PRIVATE_KEY = '0xabc123e456f789a0123456789abcdef0123456789abcdef0123456789abcdef0'; 
const TEACHER_ADDRESS     = '0x9965503B1a059419742997A47e2537357F74010C'; 
const STUDENT_ADDRESS     = '0x2111111111111111111111111111111111111111';

async function runClassDemo(){
    try{
        console.log(" --- ETHEREUM TRANSACTION CLASSROOM DEMO ---");
        console.log(`Connecting to Test Network via: ${PUBLIC_RPC}\n`);
        const nonce = await web3.eth.getTransactionCount(TEACHER_ADDRESS, 'pending');
        console.log(`Current Account Nonce (Tx Count): ${nonce}`);

        const txObject = {
            nonce: web3.utils.toHex(nonce),
            to: STUDENT_ADDRESS,
            value: web3.utils.toHex(web3.utils.toWei('0.001', 'ether')),
            gasLimit: web3.utils.toHex(21000), 
            maxPriorityFeePerGas: web3.utils.toHex(web3.utils.toWei('2', 'gwei')), 
            maxFeePerGas: web3.utils.toHex(web3.utils.toWei('30', 'gwei')),
            type: '0x2' 
        };
        console.log("Transaction Object structured successfully.");
        console.log("Signing transaction locally with the Private Key...");

        const signedTx = await web3.eth.accounts.signTransaction(txObject, TEACHER_PRIVATE_KEY);
        console.log("Transaction Signed! Cryptographic signature generated.");

        console.log("Broadcasting the signed transaction package to the peer-to-peer network...");
        const receipt = await web3.eth.sendSignedTransaction(signedTx.rawTransaction);
       
        console.log(`Success! Tx Hash: ${receipt.transactionHash}`);

    } 
    catch(error){
        console.log("\n Network Response (Validation Failed):");
        console.log(`"${error.message}"`);
        console.log("\n Lesson: The network safely verified our signature but stopped execution because the account has 0 test tokens.");
    }
}

runClassDemo();
